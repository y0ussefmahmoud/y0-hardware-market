### Step 2: Implement Direct Navigation onClick Handler

Replace the existing non-functional button HTML with an active Client Component button that reads the current window URL and navigates directly:

```tsx
'use client';

import { usePathname, useRouter } from 'next/navigation';

export function LanguageToggleBtn() {
  const pathname = usePathname();
  const router = useRouter();

  const handleToggle = (e: React.MouseEvent) => {
    e.preventDefault();
    e.stopPropagation();

    if (!pathname) return;

    // Direct segment replacement
    let targetPath = '';
    if (pathname.startsWith('/ar')) {
      targetPath = pathname.replace('/ar', '/en');
    } else if (pathname.startsWith('/en')) {
      targetPath = pathname.replace('/en', '/ar');
    } else {
      targetPath = `/ar${pathname}`;
    }

    // Set cookie to persist choice across reloads
    document.cookie = `NEXT_LOCALE=${targetPath.startsWith('/ar') ? 'ar' : 'en'}; path=/; max-age=31536000`;

    // Navigate immediately to the target route
    window.location.href = targetPath;
  };

  const isArabic = pathname?.startsWith('/ar');

  return (
    <button
      type="button"
      onClick={handleToggle}
      className="cursor-pointer flex items-center gap-1.5 px-3 py-1 rounded bg-surface-container-high hover:bg-surface-bright text-on-surface font-spec-code-sm text-spec-code-sm font-bold transition-all border border-outline-variant/30 active:scale-95 z-50"
      aria-label="Toggle Language"
    >
      <span className="material-symbols-outlined text-[16px] text-primary-container">language</span>
      <span>{isArabic ? 'English' : 'عربي'}</span>
    </button>
  );
}
```

*Note: `window.location.href` is used above as an absolute fallback to ensure a 100% guaranteed page reload to the target route if `router.push` is blocked by parent server layout bounds.*
