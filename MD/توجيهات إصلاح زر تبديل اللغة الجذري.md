### Step 2: Implement Robust Client-Side Switch Logic

Replace the existing click handler inside `LanguageSwitcher.tsx` with this tested Next.js App Router locale-switching code:

```tsx
'use client';

import { usePathname, useRouter } from 'next/navigation';

export function LanguageSwitcher() {
  const pathname = usePathname(); // e.g., "/en/shop" or "/ar/shop"
  const router = useRouter();

  const handleLanguageToggle = () => {
    if (!pathname) return;

    // Split pathname into segments: ["", "en", "shop"]
    const segments = pathname.split('/');
    const currentLocale = segments[1];

    // Determine target locale
    const newLocale = currentLocale === 'ar' ? 'en' : 'ar';

    // Replace the locale segment
    if (currentLocale === 'en' || currentLocale === 'ar') {
      segments[1] = newLocale;
    } else {
      // If root path or un-prefixed, insert target locale
      segments.splice(1, 0, newLocale);
    }

    const newPath = segments.join('/') || `/${newLocale}`;

    // Set cookie for Next.js i18n middleware persistence
    document.cookie = `NEXT_LOCALE=${newLocale}; path=/; max-age=31536000; SameSite=Lax`;

    // Force hard/soft route transition to new locale path
    router.push(newPath);
    router.refresh();
  };

  const isArabic = pathname?.startsWith('/ar');

  return (
    <button
      type="button"
      onClick={handleLanguageToggle}
      className="cursor-pointer font-spec-code-sm text-spec-code-sm text-on-surface-variant hover:text-on-surface flex items-center gap-1 transition-colors px-2 py-1 rounded bg-surface-container-high/50"
      aria-label="Toggle Language"
    >
      <span className="material-symbols-outlined text-[16px] text-primary-container">language</span>
      <span className="font-bold text-on-surface">{isArabic ? 'English' : 'عربي'}</span>
    </button>
  );
}
```
